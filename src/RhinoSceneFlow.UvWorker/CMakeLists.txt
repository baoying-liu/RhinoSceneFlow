cmake_minimum_required(VERSION 3.21)
project(SceneFlowUvWorker LANGUAGES CXX)
add_executable(SceneFlow.UvWorker main.cpp ../../third-party/xatlas/xatlas.cpp)
if(WIN32)
  enable_language(RC)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../Directory.Build.props" SCENEFLOW_PROPERTIES)
  string(REGEX MATCH "<Version>([0-9]+\\.[0-9]+\\.[0-9]+)</Version>" SCENEFLOW_MATCH "${SCENEFLOW_PROPERTIES}")
  if(NOT SCENEFLOW_MATCH)
    message(FATAL_ERROR "Missing SceneFlow three-part version")
  endif()
  set(SCENEFLOW_VERSION "${CMAKE_MATCH_1}")
  string(REPLACE "." "," SCENEFLOW_VERSION_COMMAS "${SCENEFLOW_VERSION}")
  set(SCENEFLOW_FILETYPE "VFT_APP")
  set(SCENEFLOW_DESCRIPTION "SceneFlow UV worker")
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../../installer/native/Version.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/Version.rc" @ONLY)
  target_sources(SceneFlow.UvWorker PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Version.rc")
endif()
target_compile_features(SceneFlow.UvWorker PRIVATE cxx_std_17)
target_include_directories(SceneFlow.UvWorker PRIVATE ../../third-party/xatlas)
if(MSVC)
  set_property(TARGET SceneFlow.UvWorker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(SceneFlow.UvWorker PRIVATE /O2 /EHsc)
endif()
